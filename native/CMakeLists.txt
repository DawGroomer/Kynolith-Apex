cmake_minimum_required(VERSION 3.20)
project(kynolith_lmu_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(kynolith-lmu-bridge main.cpp)
if(NOT DEFINED LMU_SDK_DIR)
  message(FATAL_ERROR "Pass -DLMU_SDK_DIR=<LMU>/Support/SharedMemoryInterface")
endif()
target_include_directories(kynolith-lmu-bridge PRIVATE "${LMU_SDK_DIR}")
target_compile_definitions(kynolith-lmu-bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
